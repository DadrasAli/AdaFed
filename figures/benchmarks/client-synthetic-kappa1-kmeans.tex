\newcommand{\LegKey}[1]{\tikz[baseline=-0.55ex]{\draw[#1] (0,0)--(0.52,0);}}
\newcommand{\LegKeyLast}{%
  \tikz[baseline=-0.55ex]{%
    \draw[black!75,densely dashed,line width=0.62pt] (0,0)--(0.52,0);
    \fill[black!75] (0.26,0) circle (0.7pt);}}
\newcommand{\LegSep}{\hspace{0.8em}}

% Asymmetric one-sided standard-deviation bands around stochastic means.
% The table bounds are computed directionally, so the lower edge remains
% nonnegative without an arbitrary clipping threshold.
\newcommand{\MeanBand}[3]{%
  \addplot[name path=#1low,draw=none,no marks,forget plot]
    table[x=round,y=random_low] {#3};
  \addplot[name path=#1high,draw=none,no marks,forget plot]
    table[x=round,y=random_high] {#3};
  \addplot[fill=#2,fill opacity=0.22,draw=none,forget plot]
    fill between[of=#1low and #1high];
}
\newcommand{\LastBand}[3]{%
  \addplot[name path=#1low,draw=none,no marks,forget plot]
    table[x=round,y=last_low] {#3};
  \addplot[name path=#1high,draw=none,no marks,forget plot]
    table[x=round,y=last_high] {#3};
  \addplot[fill=#2,fill opacity=0.17,draw=none,forget plot]
    fill between[of=#1low and #1high];
}
